## 14.1 测试策略概述

### 14.1.1 测试金字塔

```
         /\
        /  \
       / E2E \         端到端测试 (5%)
      /_______\
     /         \
    /  Integration\    集成测试 (15%)
   /_______________\
  /                 \
 /    Unit Tests     \  单元测试 (80%)
/_____________________\
```

### 14.1.2 测试覆盖目标

| 测试类型 | 覆盖率目标 | 说明 |
|---------|-----------|------|
| **单元测试** | > 70% | Service、Converter、Utils |
| **集成测试** | > 50% | API 接口、数据库操作 |
| **端到端测试** | 核心流程 | 用户登录、权限分配 |

### 14.2.2 单元测试示例

**Converter 测试：**

```java
@ExtendWith(MockitoExtension.class)
class ISysUserDTOConvertTest {
    
    @Test
    void fromCreateDTO() {
        // 准备数据
        SysUserCreateDTO dto = new SysUserCreateDTO();
        dto.setRealName("张三");
        dto.setMobile("13800138000");
        dto.setLoginName("zhangsan");
        dto.setPassword("123456");
        
        // 执行转换
        SysUserBO bo = ISysUserDTOConvert.INSTANCE.fromCreateDTO(dto);
        
        // 断言
        assertThat(bo).isNotNull();
        assertThat(bo.getRealName()).isEqualTo("张三");
        assertThat(bo.getMobile()).isEqualTo("13800138000");
        assertThat(bo.getLoginName()).isEqualTo("zhangsan");
    }
    
    @Test
    void toVO() {
        // 准备数据
        SysUserBO bo = new SysUserBO();
        bo.setId(1);
        bo.setRealName("张三");
        bo.setStatus("1");
        
        // 执行转换
        SysUserVO vo = ISysUserDTOConvert.INSTANCE.toVO(bo);
        
        // 断言
        assertThat(vo.getId()).isEqualTo(1);
        assertThat(vo.getRealName()).isEqualTo("张三");
        assertThat(vo.getStatus()).isEqualTo("1");
    }
}
```

**Service 单元测试：**

```java
@ExtendWith(MockitoExtension.class)
class SysUserAppServiceTest {
    
    @Mock
    private ISysUserService sysUserService;
    
    @Mock
    private ISysUserAuthsService sysUserAuthsService;
    
    @Mock
    private ISysOrgService sysOrgService;
    
    @InjectMocks
    private SysUserAppService sysUserAppService;
    
    @Test
    void save_当登录账号已存在时返回失败() {
        // 准备数据
        SysUserCreateDTO dto = new SysUserCreateDTO();
        dto.setLoginName("zhangsan");
        dto.setPassword("123456");
        
        SysUserAuthsVO existingAuth = new SysUserAuthsVO();
        existingAuth.setId(1);
        
        // Mock 行为
        when(sysUserAuthsReadModelService.getOne(any()))
            .thenReturn(existingAuth);
        
        // 执行
        R<SysUserVO> result = sysUserAppService.save(dto);
        
        // 断言
        assertThat(result.getCode()).isNotEqualTo(0);
        assertThat(result.getMsg()).contains("已存在");
        
        // 验证交互
        verify(sysUserAuthsReadModelService).getOne(any());
        verify(sysUserService, never()).save(any());
    }
    
    @Test
    void save_当登录账号可用时创建成功() {
        // 准备数据
        SysUserCreateDTO dto = new SysUserCreateDTO();
        dto.setLoginName("zhangsan");
        dto.setPassword("123456");
        dto.setRealName("张三");
        
        SysUserBO savedUser = new SysUserBO();
        savedUser.setId(1);
        savedUser.setLoginName("zhangsan");
        
        SysOrgBO org = new SysOrgBO();
        org.setId(1);
        org.setOrgCode("PXC001");
        
        // Mock 行为
        when(sysUserAuthsReadModelService.getOne(any())).thenReturn(null);
        when(sysOrgService.getById(any())).thenReturn(org);
        when(sysUserService.save(any())).thenReturn(savedUser);
        when(sysUserAuthsService.save(any())).thenReturn(true);
        
        // 执行
        R<SysUserVO> result = sysUserAppService.save(dto);
        
        // 断言
        assertThat(result.getCode()).isEqualTo(0);
        assertThat(result.getData()).isNotNull();
        assertThat(result.getData().getId()).isEqualTo(1);
        
        // 验证交互
        verify(sysUserService).save(any(SysUserBO.class));
        verify(sysUserAuthsService).save(any(SysUserAuthsBO.class));
    }
}
```

## 14.3 集成测试

### 14.3.1 API 接口测试

```java
@SpringBootTest
@AutoConfigureMockMvc
class SysUserApiTest {
    
    @Autowired
    private MockMvc mockMvc;
    
    @Autowired
    private ObjectMapper objectMapper;
    
    @Test
    void testPageQuery() throws Exception {
        // 准备请求
        SysUserPageQueryDTO queryDTO = new SysUserPageQueryDTO();
        queryDTO.setPageNo(1);
        queryDTO.setPageSize(10);
        
        // 执行请求
        mockMvc.perform(post("/system/v1/sys-user/page")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(queryDTO)))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(0))
            .andExpect(jsonPath("$.data.list").isArray())
            .andExpect(jsonPath("$.data.total").exists());
    }
    
    @Test
    void testGetById() throws Exception {
        // 执行请求
        mockMvc.perform(get("/system/v1/sys-user/get")
                .param("id", "1"))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(0))
            .andExpect(jsonPath("$.data.id").value(1));
    }
    
    @Test
    void testSave_未登录时返回认证失败() throws Exception {
        // 准备数据
        SysUserCreateDTO dto = new SysUserCreateDTO();
        dto.setLoginName("test");
        dto.setPassword("123456");
        
        // 执行请求（无 Token）
        mockMvc.perform(post("/system/v1/sys-user")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(301)); // 未登录
    }
    
    @Test
    void testSave_有权限时创建成功() throws Exception {
        // 准备数据
        SysUserCreateDTO dto = new SysUserCreateDTO();
        dto.setLoginName("testuser");
        dto.setPassword("123456");
        dto.setRealName("测试用户");
        
        String token = "Bearer test_token";
        
        // 执行请求
        mockMvc.perform(post("/system/v1/sys-user")
                .contentType(MediaType.APPLICATION_JSON)
                .header("Authorization", token)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isOk())
            .andExpect(jsonPath("$.code").value(0));
    }
}
```

### 14.3.2 数据库集成测试

```java
@SpringBootTest
@AutoConfigureTestDatabase(replace = AutoConfigureTestDatabase.Replace.ANY)
class SysUserRepositoryTest {
    
    @Autowired
    private SysUserMapper sysUserMapper;
    
    @Autowired
    private TransactionTemplate transactionTemplate;
    
    @BeforeEach
    void setUp() {
        // 清理测试数据
        sysUserMapper.delete(new LambdaQueryWrapper<SysUserPO>());
    }
    
    @Test
    void testInsert() {
        // 准备数据
        SysUserPO user = new SysUserPO();
        user.setRealName("测试用户");
        user.setMobile("13800138000");
        user.setStatus("1");
        
        // 执行插入
        int rows = sysUserMapper.insert(user);
        
        // 断言
        assertThat(rows).isEqualTo(1);
        assertThat(user.getId()).isNotNull();
    }
    
    @Test
    void testUpdate() {
        // 插入测试数据
        SysUserPO user = new SysUserPO();
        user.setRealName("原名称");
        sysUserMapper.insert(user);
        
        // 更新
        user.setRealName("新名称");
        int rows = sysUserMapper.updateById(user);
        
        // 断言
        assertThat(rows).isEqualTo(1);
        
        // 验证
        SysUserPO updated = sysUserMapper.selectById(user.getId());
        assertThat(updated.getRealName()).isEqualTo("新名称");
    }
    
    @Test
    void testDelete() {
        // 插入测试数据
        SysUserPO user = new SysUserPO();
        sysUserMapper.insert(user);
        
        // 删除
        int rows = sysUserMapper.deleteById(user.getId());
        
        // 断言
        assertThat(rows).isEqualTo(1);
        assertThat(sysUserMapper.selectById(user.getId())).isNull();
    }
    
    @Test
    void testPageQuery() {
        // 插入测试数据
        for (int i = 0; i < 15; i++) {
            SysUserPO user = new SysUserPO();
            user.setRealName("用户" + i);
            sysUserMapper.insert(user);
        }
        
        // 分页查询
        Page<SysUserPO> page = sysUserMapper.selectPage(
            Page.of(1, 10), 
            new LambdaQueryWrapper<SysUserPO>()
        );
        
        // 断言
        assertThat(page.getRecords()).hasSize(10);
        assertThat(page.getTotal()).isEqualTo(15);
    }
}
```

### 14.4.1 用户创建流程测试

```java
@SpringBootTest
class UserCreationIntegrationTest {
    
    @Autowired
    private SysUserAppService sysUserAppService;
    
    @Test
    void testCompleteUserCreation() {
        // 1. 准备创建数据
        SysUserCreateDTO dto = new SysUserCreateDTO();
        dto.setLoginName("newuser");
        dto.setPassword("123456");
        dto.setRealName("新用户");
        dto.setMobile("13800138000");
        dto.setOrgCode("PXC001");
        dto.setPostCode("DEV001");
        
        // 2. 执行创建
        R<SysUserVO> result = sysUserAppService.save(dto);
        
        // 3. 验证创建成功
        assertThat(result.getCode()).isEqualTo(0);
        assertThat(result.getData()).isNotNull();
        
        Integer userId = result.getData().getId();
        
        // 4. 验证用户信息
        R<SysUserVO> getUserResult = sysUserAppService.getById(userId);
        assertThat(getUserResult.getData().getLoginName()).isEqualTo("newuser");
        assertThat(getUserResult.getData().getRealName()).isEqualTo("新用户");
        
        // 5. 验证登录凭证创建
        // 6. 验证组织关联创建
        // 7. 验证岗位关联创建
    }
}
```

## 14.9 测试最佳实践

### 14.9.1 测试命名规范

```java
// 方法命名：场景_预期结果
@Test
void save_当登录账号已存在时返回失败() { }

@Test
void save_当登录账号可用时创建成功() { }

@Test

```

### 14.9.2 AAA 模式

```java
@Test
void testSave() {
    // Arrange - 准备
    SysUserCreateDTO dto = new SysUserCreateDTO();
    dto.setLoginName("test");
    
    // Act - 执行
    R<SysUserVO> result = sysUserAppService.save(dto);
    
    // Assert - 断言
    assertThat(result.getCode()).isEqualTo(0);
}
```

### 14.9.3 测试隔离
